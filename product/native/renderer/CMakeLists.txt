cmake_minimum_required(VERSION 3.21)
project(Flamoris2D.Renderer.Native LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The selected D3D11 renderer requires Windows")
endif()
add_library(Flamoris2D.Renderer.Native SHARED src/d3d11_renderer.cpp)
target_compile_features(Flamoris2D.Renderer.Native PRIVATE cxx_std_17)
target_compile_definitions(Flamoris2D.Renderer.Native PRIVATE FL2DR_BUILD NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(Flamoris2D.Renderer.Native PUBLIC include PRIVATE ../core/third_party/picojson)
target_link_libraries(Flamoris2D.Renderer.Native PRIVATE d3d11 d3dcompiler)
if(MSVC)
  target_compile_options(Flamoris2D.Renderer.Native PRIVATE /W4 /WX /utf-8)
endif()
